🎨 SOLUÇÃO PARA LOVABLE - O QUE VOCÊ PODE FAZER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

LIMITAÇÕES DO LOVABLE:
❌ Não pode rodar npm run dev localmente
❌ Não pode acessar terminal
❌ Não pode fazer build manual
✅ PODE editar código no editor do Lovable
✅ PODE executar SQL no Supabase Dashboard
✅ PODE ver console do navegador (F12)
✅ PODE fazer deploy automático

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
SOLUÇÃO ADAPTADA PARA LOVABLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PASSO 1: VERIFICAR DADOS NO SUPABASE (5 minutos)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Abra o Supabase Dashboard
2. Vá em SQL Editor
3. Execute esta query:

SELECT 
  title,
  plan_snapshot->'1500'->0->'meals'->0 as primeira_refeicao
FROM v3_diet_templates
WHERE title = 'Emagrecimento 1500 kcal';

4. Me diga o resultado (copie e cole aqui)


PASSO 2: AGUARDAR DEPLOY AUTOMÁTICO DO LOVABLE (2 minutos)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

O Lovable faz deploy automático quando você salva o código.

1. Aguarde a mensagem "Deploy successful" no Lovable
2. Aguarde 1-2 minutos para o cache limpar
3. Recarregue a página do app (Ctrl+Shift+R)


PASSO 3: TESTAR NO NAVEGADOR (3 minutos)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Abra o app do Lovable (URL do seu projeto)
2. Abra o Console (F12) - aba "Console"
3. Faça login
4. Vá no Editor V3
5. Clique em "Biblioteca"
6. Selecione "Emagrecimento 1500 kcal"
7. Escolha perfil "1500 kcal"
8. Clique em "Plotar Template"
9. OLHE O CONSOLE e me diga o que aparece
